getDealsFromHistoryTable

 Select  
 Upper(CUS.CUST_B_NAME) as CUST_B_NAME, 
 PRPS.BBTP_B_DESC,  
 SUBSTR(PRPS.BBTP_B_DESC, -3, 3) || '-' || HSMM_DEAL_NUM_3 as DealRef,    
 LMT.CPLM_BB_EXPOSURE_LMT_AMNT,  
 LMT.CPLM_PB_EXPOSURE_LMT_AMNT ,   
 HSMM_AMNT as AMNT,   
 HSMM_INT_RATE as INT_RATE,   
 HSMM_INTEREST as INTEREST,   
 HSMM_VALUE_DATE as VALUE_DATE,    
 HSMM_MATURITY_DATE as MATURITY_DATE,   
 HSMM_OPER_DATE as OPER_DATE,   
 HSMM_DEAL_NUM_3 as DEAL_NUM_3 ,  
 'HS' as xTable,  
 TRDL.STATE as STATE,  
 TRST.DESCRIPTION as STATEDesc, 
 TRST.SUBTITLE as SUBTITLE, 
 CASE TRDL.STATE 
   WHEN '1000' THEN 'YES'  
   WHEN '9999' THEN 'YES'  
   ELSE 'NO' 
   END as isMarkedInMV,  
 TRST.COLOR as BoxColor,   
 TRST.FontCOLOR as FontColor,   
 TRDL.RESTRICTEDTO as RESTRICTEDTO,  
 TRDL.ISDISABLED as ISDISABLED,   
 (HSMM_INTEREST + HSMM_AMNT) as AmountatMaturity,   
 TRDL.DATETIME as EntryDate, 
 '0.0' as AMOUNTSALE, 
 '0.0' as EXCHANGE_RATE, 
 'N' as IS_FX, 
 CURR.CURR_ISO as Currency, 
 CURR.CURR_DEC_NUM as DecDgts, 
 TRDL.BBTP_CODE as BBTP_CODE 
 from BIMM_HS_MONEY_MARKET@ICBS  MM   
 inner join BBSD_CUSTOMERs@ICBS  CUS on MM.CUST_ID = CUS.CUST_ID   
 inner join BBSD_TREASURY_PURPOSES@ICBS  PRPS on MM.BBTP_CODE = PRPS.BBTP_CODE   
 inner join BBSD_COUNTERPARTY_LIMITS@ICBS  LMT on MM.CUST_ID = LMT.CUST_ID    
 inner Join TRSRY_DEALS TRDL on TRDL.MVMM_DEAL_NUM_3 = MM.HSMM_DEAL_NUM_3    
 inner Join TRSRY_STATES TRST on TRDL.STATE = TRST.STATE    
 inner Join BBsc_currencies@ICBS  CURR on CURR.CURR_CODE = MM.CURR_CODE   
   
 where  
 Not((MM.BBTP_CODE = 113 and MM.HSMM_BRCH = 850) or (MM.BBTP_CODE = 115 and MM.HSMM_BRCH = 850))
 order by HSMM_DEAL_NUM_3 desc 
 fetch first 300 rows only

  mm.MVMM_OPER_DATE As OPER_DATE,    
  mm.MVMM_DEAL_NUM_3 as DEAL_NUM_3 ,  
  'MV' as xTable,  
 AMMT.STATE,  
  TRST.DESCRIPTION as STATEDesc,

  TRST.FontCOLOR as FontColor,   
  AMMT.RESTRICTEDTO as RESTRICTEDTO,  
  AMMT.ISDISABLED as ISDISABLED,  
  (mm.MVMM_INTEREST+mm.MVMM_AMNT) as AmountatMaturity,

 inner Join BBSD_COUNTERPARTY_LIMITS@ICBS  LMT on MM.CUST_ID = LMT.CUST_ID   
 inner Join TRSRY_DEALS AMMT on MM.MVMM_DEAL_NUM_3 = AMMT.MVMM_DEAL_NUM_3   
 inner Join TRSRY_STATES TRST on TRST.STATE = AMMT.STATE   
 inner Join BBsc_currencies@ICBS CURR on CURR.CURR_CODE = MM.CURR_CODE    where    MM.MVMM_DEAL_NUM_3 in ((Select MVMM_DEAL_NUM_3 from TRSRY_DEALS where STATE not in ('0000')  
 Union Select MVMM_DEAL_NUM_3 from TRSRY_DEALS AB  LEFT JOIN GBSS_USERS_DEFS@ICBS  U2 ON AB.MVMM_INPUT_USER_HO = U2.UDEF_ID   inner Join BBsc_currencies@ICBS  CURR on CURR.CURR_CODE = MM.CURR_CODE    where AB.STATE='0000' and U2.UDEF_CODE like '%2271%')) 
 and   Not ((MM.BBTP_CODE = 113 and MM.MVMM_BRCH = 850) or (MM.BBTP_CODE = 115 and MM.MVMM_BRCH = 850))  
getDealsFromHistoryNotExistedIn_TRSRY_DEALS

 PRPS.BBTP_B_DESC, 
 SUBSTR(PRPS.BBTP_B_DESC, -3, 3) || '-' || HSMM_DEAL_NUM_3 as DealRef,

 LMT.CPLM_PB_EXPOSURE_LMT_AMNT ,  
 HSMM_AMNT as AMNT,  
 HSMM_INT_RATE as INT_RATE,  
 HSMM_INTEREST as INTEREST,  
 HSMM_VALUE_DATE as VALUE_DATE,   
 HSMM_MATURITY_DATE as MATURITY_DATE,  
 HSMM_OPER_DATE as OPER_DATE,  
 HSMM_DEAL_NUM_3 as DEAL_NUM_3 , 
 'HS' as xTable, 
 '1000' as STATE, 
 'Matured' as STATEDesc,
 TRST.SUBTITLE as SUBTITLE, 
 'YES' isMarkedInMV, 
 TRST.COLOR as BoxColor,  
 TRST.FontCOLOR as FontColor,  
 '*' as RESTRICTEDTO, 
 'NO' as ISDISABLED,  
 (HSMM_INTEREST + HSMM_AMNT) as AmountatMaturity,  
 to_char(MM.TIME_STAMP) as EntryDate, 
 '0.0' as AMOUNTSALE, 
 '0.0' as EXCHANGE_RATE, 
 'N' as IS_FX, 
 CURR.CURR_ISO as Currency, 
 CURR.CURR_DEC_NUM as DecDgts,

 inner join BBSD_TREASURY_PURPOSES@ICBS  PRPS on MM.BBTP_CODE = PRPS.BBTP_CODE  
 inner join BBSD_COUNTERPARTY_LIMITS@ICBS  LMT on MM.CUST_ID = LMT.CUST_ID   
 inner Join TRSRY_STATES TRST on TRST.STATE='1000' 
 inner Join BBsc_currencies@ICBS  CURR on CURR.CURR_CODE = MM.CURR_CODE   
 where Not ((MM.BBTP_CODE = 113 and MM.HSMM_BRCH = 850) or (MM.BBTP_CODE = 115 and MM.HSMM_BRCH = 850))  
  and HSMM_DEAL_NUM_3 Not in (Select MVMM_DEAL_NUM_3 from TRSRY_DEALS)  
 order by HSMM_DEAL_NUM_3 desc
